## Install your Engine

[Create your Threadify account](https://threadify.dev/signup) to get a license
key, then install the Engine:

```bash
curl --fail --show-error --location --proto '=https' --proto-redir '=https' \
  https://github.com/creativeJoe007/ThreadifyEngine/releases/latest/download/install.sh \
  --output install.sh
sh install.sh
```

Edit the installed `config.yaml` with your license key, PostgreSQL connection
and hash-chain secret, then run the start command printed by the installer.
Open `http://localhost:8081` to sign in to your dashboard. See the
[self-hosting guide](https://github.com/creativeJoe007/ThreadifyEngine/blob/main/threadify-go/SELF_HOSTING.md)
for the full setup and [authentication](/authentication) for browser sign-in.

## 2. Connect to Threadify

<Note>
  Need an API key? [Sign in to your Engine](/authentication), then open **Developer → API Keys** to create one.
</Note>

<Tabs>
<Tab title="JavaScript">
```javascript
import { Threadify } from '@threadify/sdk';

// Connect with your API key and service name
const connection = await Threadify.connect('your-api-key', 'my-service', {
  engineUrl: 'https://threadify.example.com' // Your Settings → Engine address
});
```
</Tab>
<Tab title="Go">
```go
ctx := context.Background()
conn, err := threadify.Connect(ctx, "your-api-key", 
    threadify.WithServiceName("my-service"),
    threadify.WithEngineURL("https://threadify.example.com"),
)
if err != nil {
    log.Fatal(err)
}
defer conn.Close()
```
</Tab>
<Tab title="Python">
```python
from threadify import Threadify

# Connect with your API key and service name
connection = await Threadify.connect(
    "your-api-key",
    service_name="my-service",
    engine_url="https://threadify.example.com",
)
```
</Tab>
</Tabs>

## 3. Create Your First Thread

<Tabs>
<Tab title="JavaScript">
```javascript
// Start tracking a workflow with a label
const thread = await connection.start('Order-123');

// Or start with a contract and a label
const thread = await connection.start('Order-789', 'order_fulfillment');
```
</Tab>
<Tab title="Go">
```go
// Start tracking a workflow with a label
thread, err := conn.Start(ctx, "Order-123")
if err != nil {
    log.Fatal(err)
}

// Or start with a contract and a label
thread, err = conn.Start(ctx, "Order-789", threadify.WithContract("order_fulfillment"))
if err != nil {
    log.Fatal(err)
}
```
</Tab>
<Tab title="Python">
```python
# Start tracking a workflow with a label
thread = await connection.start("Order-123")

# Or start with a contract and a label
thread = await connection.start("Order-789", contract_name="order_fulfillment")
```
</Tab>
</Tabs>

## 4. Add Steps to Your Process

<Tabs>
<Tab title="JavaScript">
```javascript
// Create step first, then call methods on it
const step = thread.step('validate_input');
await step.addContext({ userId: 'user-123' }).success();

// Or chain directly on the step builder
await thread.step('process_data')
  .addContext({ records: 42 })
  .success();
```
</Tab>
<Tab title="Go">
```go
// Create step first, then call methods on it
step := thread.Step("validate_input")
_, err = step.AddContext(map[string]any{"userId": "user-123"}).Success(ctx)
if err != nil {
    log.Fatal(err)
}

// Or chain directly on the step builder
_, err = thread.Step("process_data").
    AddContext(map[string]any{"records": 42}).
    Success(ctx)
if err != nil {
    log.Fatal(err)
}
```
</Tab>
<Tab title="Python">
```python
# Create step first, then call methods on it
step = thread.step("validate_input")
await step.add_context({"userId": "user-123"}).success()

# Or chain directly on the step builder
await (
    thread.step("process_data")
    .add_context({"records": 42})
    .success()
)
```
</Tab>
</Tabs>
